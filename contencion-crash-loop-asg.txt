# CONTENCION CRASH LOOP ASG - intelisrcpa prd
# Objetivo: detener el bucle de reemplazo por "ELB system health check failure"
#           suspendiendo HealthCheck y ReplaceUnhealthy en los 7 ASG, dejando
#           las instancias vivas para diagnosticar.
# Region:   us-east-1  (ajustar si aplica)
# Ejecutar en: AWS CloudShell / consola web (NO toca el ALB; es reversible)
###############################################################################


###############################################################################
# PASO 1 - Confirmar los nombres exactos de los ASG (ejecutar y revisar la lista)
###############################################################################
aws autoscaling describe-auto-scaling-groups \
  --query "AutoScalingGroups[?contains(AutoScalingGroupName, 'intelisrcpa') && contains(AutoScalingGroupName, 'web')].AutoScalingGroupName" \
  --output text --region us-east-1


###############################################################################
# PASO 2 - Suspender HealthCheck y ReplaceUnhealthy en TODOS los ASG
#          (detiene el bucle de reemplazo)
###############################################################################
for asg in $(aws autoscaling describe-auto-scaling-groups \
  --query "AutoScalingGroups[?contains(AutoScalingGroupName, 'intelisrcpa') && contains(AutoScalingGroupName, 'web')].AutoScalingGroupName" \
  --output text --region us-east-1); do
  echo "Suspendiendo en $asg ..."
  aws autoscaling suspend-processes \
    --auto-scaling-group-name "$asg" \
    --scaling-processes HealthCheck ReplaceUnhealthy \
    --region us-east-1
done


###############################################################################
# PASO 3 - Verificar que quedaron suspendidos
###############################################################################
aws autoscaling describe-auto-scaling-groups \
  --query "AutoScalingGroups[?contains(AutoScalingGroupName, 'intelisrcpa') && contains(AutoScalingGroupName, 'web')].{Name:AutoScalingGroupName,Suspended:SuspendedProcesses[].ProcessName}" \
  --output table --region us-east-1
