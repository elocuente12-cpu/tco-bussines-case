# 1. Ver si hay una GPO aplicando timezone
gpresult /r /scope:computer | Select-String -Pattern "Applied|Time"

# 2. Ver el historial de cambios de la zona (Event Log)
Get-WinEvent -FilterHashtable @{LogName='System';ProviderName='Microsoft-Windows-Kernel-General'} -MaxEvents 20 |
  Where-Object {$_.Message -like "*time zone*"} | Format-List TimeCreated, Message

# 1. Ver qué tareas programadas corren pwsh/powershell y podrían tocar timezone
Get-ScheduledTask | Where-Object {$_.Actions.Execute -match "pwsh|powershell"} |
  Select-Object TaskName, TaskPath, State

# 2. Buscar cualquier script que mencione timezone/Eastern en el sistema
Get-ChildItem C:\ -Recurse -Include *.ps1 -ErrorAction SilentlyContinue |
  Select-String -Pattern "Eastern Standard Time|Set-TimeZone|tzutil" -ErrorAction SilentlyContinue |
  Select-Object Path, LineNumber, Line

# 3. Confirmar la zona actual y forzarla de vuelta para probar
tzutil /g



# 1. Tareas programadas que usan pwsh (PowerShell 7)
Get-ScheduledTask | Where-Object {($_.Actions | ForEach-Object Execute) -match "pwsh"} |
  Select-Object TaskName, TaskPath | Format-Table -Auto
