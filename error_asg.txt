# 1. Zona a nivel de SISTEMA (lo que aplica a servicios/IIS)
tzutil /g
# Esperado: "SA Pacific Standard Time"

# 2. La hora que ve el sistema vs UTC
Get-Date
[DateTime]::UtcNow
# Si Get-Date = UTC - 5h => sistema correcto

# 3. Lo que ve TU sesion RDP (puede diferir)
[System.TimeZoneInfo]::Local.Id

# 1. Ver si hay una GPO aplicando timezone
gpresult /r /scope:computer | Select-String -Pattern "Applied|Time"

# 2. Ver el historial de cambios de la zona (Event Log)
Get-WinEvent -FilterHashtable @{LogName='System';ProviderName='Microsoft-Windows-Kernel-General'} -MaxEvents 20 |
  Where-Object {$_.Message -like "*time zone*"} | Format-List TimeCreated, Message

# 3. Ver todas las GPOs aplicadas al equipo
gpresult /h C:\gpreport.html   # y revisar el HTML, buscar "Time Zone"
